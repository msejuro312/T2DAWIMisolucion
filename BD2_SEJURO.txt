CREATE DATABASE BD2_SEJURO;
USE BD_SEJURO;

CREATE OR ALTER TABLE tipo_habitacion (
	cod_tipo INT PRIMARY KEY AUTO INCREMENT,
	nom_tipo_habitacion VARCHAR(100) NOT NULL,
	precio DECIMAL (10,2) NOT NULL
);

CREATE OR ALTER TABLE cliente (
	id_cliente INT PRIMARY KEY AUTO INCREMENT,
	nom_cliente VARCHAR(100) NOT NULL,
	);

CREATE OR ALTER TABLE reserva (
	num_reserva INT PRIMARY KEY AUTO INCREMENT,
	fecha_reserva DATE NOT NULL,
	id_cliente INT NOT NULL,
	cod_tipo INT NOT NULL,
	estado CHAR(1) NOT NULL,
	FOREIGN KEY (id_cliente) REFERENCES cliente (id_cliente),
	FOREIGN KEY (cod_tipo) REFERENCES tipo_habitacion(cod_tipo),
	);
